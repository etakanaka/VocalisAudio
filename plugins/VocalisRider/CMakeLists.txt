cmake_minimum_required(VERSION 3.22)
project(VocalisRiderLive VERSION 1.0.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(JUCE_PATH "$ENV{HOME}/AudioDevelopment/JUCE" CACHE PATH "JUCE source tree")
set(VOCALIS_SANITIZER "" CACHE STRING "Isolated tests: ADDRESS or THREAD; leave empty for release")
if(VOCALIS_SANITIZER STREQUAL "ADDRESS")
 add_compile_options(-fsanitize=address,undefined,float-cast-overflow -fno-omit-frame-pointer)
 add_link_options(-fsanitize=address,undefined,float-cast-overflow)
elseif(VOCALIS_SANITIZER STREQUAL "THREAD")
 add_compile_options(-fsanitize=thread -fno-omit-frame-pointer)
 add_link_options(-fsanitize=thread)
elseif(NOT VOCALIS_SANITIZER STREQUAL "")
 message(FATAL_ERROR "Unknown VOCALIS_SANITIZER")
endif()
add_subdirectory("${JUCE_PATH}" JUCE EXCLUDE_FROM_ALL)
juce_add_plugin(VocalisRiderLive COMPANY_NAME "Vocalis Audio" BUNDLE_ID "audio.vocalis.riderlive" PLUGIN_MANUFACTURER_CODE Vcls PLUGIN_CODE VrLv FORMATS VST3 PRODUCT_NAME "Vocalis Rider Live" IS_SYNTH FALSE NEEDS_MIDI_INPUT FALSE NEEDS_MIDI_OUTPUT FALSE COPY_PLUGIN_AFTER_BUILD FALSE)
target_sources(VocalisRiderLive PRIVATE Source/Plugin.cpp Source/Editor.cpp)
target_compile_definitions(VocalisRiderLive PUBLIC JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0 JUCE_VST3_CAN_REPLACE_VST2=0)
target_link_libraries(VocalisRiderLive PRIVATE juce::juce_audio_utils PUBLIC juce::juce_recommended_config_flags)
enable_testing()
add_executable(dsp_tests tests/DspTests.cpp)
target_include_directories(dsp_tests PRIVATE Source)
add_test(NAME DSP COMMAND dsp_tests)
add_executable(integration_tests tests/Integration.cpp)
target_include_directories(integration_tests PRIVATE Source "${JUCE_PATH}/modules")
target_compile_definitions(integration_tests PRIVATE JUCE_GLOBAL_MODULE_SETTINGS_INCLUDED=1 JUCE_MODAL_LOOPS_PERMITTED=1 JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0)
target_link_libraries(integration_tests PRIVATE VocalisRiderLive)
add_test(NAME Integration COMMAND integration_tests)
set_tests_properties(Integration PROPERTIES WORKING_DIRECTORY "${CMAKE_SOURCE_DIR}")
target_sources(integration_tests PRIVATE tests/Allocation.cpp)
target_link_options(integration_tests PRIVATE -Wl,--wrap=malloc -Wl,--wrap=calloc -Wl,--wrap=realloc)
add_executable(vst3_smoke tests/Vst3Smoke.cpp)
target_include_directories(vst3_smoke PRIVATE "${JUCE_PATH}/modules/juce_audio_processors_headless/format_types/VST3_SDK")
target_compile_definitions(vst3_smoke PRIVATE RELEASE=1)
target_link_libraries(vst3_smoke PRIVATE dl)
add_test(NAME VST3_Module COMMAND vst3_smoke "${CMAKE_BINARY_DIR}/VocalisRiderLive_artefacts/Release/VST3/Vocalis Rider Live.vst3/Contents/x86_64-linux/Vocalis Rider Live.so")

set_source_files_properties(Source/Plugin.cpp Source/Editor.cpp PROPERTIES COMPILE_OPTIONS "-Wall;-Wextra;-Wpedantic")
function(vocalis_test target source)
 add_executable(${target} ${source})
 target_include_directories(${target} PRIVATE Source "${JUCE_PATH}/modules")
 target_compile_definitions(${target} PRIVATE JUCE_GLOBAL_MODULE_SETTINGS_INCLUDED=1 JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0)
 target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic)
 target_link_libraries(${target} PRIVATE VocalisRiderLive)
endfunction()
vocalis_test(audit_tests tests/Audit.cpp)
add_test(NAME Audit COMMAND audit_tests --final "${CMAKE_SOURCE_DIR}/tests/fixtures/pre-debug/legacy-state.bin")
set_tests_properties(Audit PROPERTIES WORKING_DIRECTORY "${CMAKE_SOURCE_DIR}")
vocalis_test(gui_tests tests/GuiTests.cpp)
add_test(NAME GUI COMMAND gui_tests)
set_tests_properties(GUI PROPERTIES WORKING_DIRECTORY "${CMAKE_SOURCE_DIR}")
vocalis_test(performance_tests tests/Performance.cpp)
target_sources(performance_tests PRIVATE tests/Allocation.cpp)
target_link_options(performance_tests PRIVATE -Wl,--wrap=malloc -Wl,--wrap=calloc -Wl,--wrap=realloc)
add_executable(dsp_audit tests/DspAudit.cpp)
target_include_directories(dsp_audit PRIVATE Source)
target_compile_options(dsp_audit PRIVATE -Wall -Wextra -Wpedantic)
add_test(NAME DSP_Audit COMMAND dsp_audit)
vocalis_test(thread_stress tests/ThreadStress.cpp)
add_test(NAME Thread_Stress COMMAND thread_stress)
find_package(Python3 COMPONENTS Interpreter REQUIRED)
add_test(NAME Install_Safety COMMAND "${Python3_EXECUTABLE}" "${CMAKE_SOURCE_DIR}/tests/InstallTests.py")
add_test(NAME VST3_Reload COMMAND vst3_smoke "${CMAKE_BINARY_DIR}/VocalisRiderLive_artefacts/Release/VST3/Vocalis Rider Live.vst3/Contents/x86_64-linux/Vocalis Rider Live.so" 20)
