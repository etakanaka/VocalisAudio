cmake_minimum_required(VERSION 3.22)
project(VocalisVoice VERSION 1.0.0 LANGUAGES C CXX)
if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux" OR NOT CMAKE_SYSTEM_PROCESSOR MATCHES "^(x86_64|AMD64)$")
    message(FATAL_ERROR "Vocalis Voice targets native Linux x86_64 only")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(JUCE_PATH "$ENV{HOME}/AudioDevelopment/JUCE" CACHE PATH "Existing JUCE installation")
add_subdirectory("${JUCE_PATH}" JUCE EXCLUDE_FROM_ALL)
juce_add_plugin(VocalisVoice
 COMPANY_NAME "Vocalis Audio" BUNDLE_ID "audio.vocalis.voice"
 PLUGIN_MANUFACTURER_CODE Vcls PLUGIN_CODE Vvce
 FORMATS VST3 PRODUCT_NAME "Vocalis Voice"
 IS_SYNTH FALSE NEEDS_MIDI_INPUT FALSE NEEDS_MIDI_OUTPUT FALSE
 IS_MIDI_EFFECT FALSE EDITOR_WANTS_KEYBOARD_FOCUS FALSE
 VST3_CATEGORIES Fx Dynamics Reverb
 COPY_PLUGIN_AFTER_BUILD FALSE)
target_sources(VocalisVoice PRIVATE src/PluginProcessor.cpp src/PluginEditor.cpp)
target_include_directories(VocalisVoice PUBLIC src)
target_compile_definitions(VocalisVoice PUBLIC JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0 JUCE_VST3_CAN_REPLACE_VST2=0)
target_link_libraries(VocalisVoice PRIVATE juce::juce_audio_utils juce::juce_dsp PUBLIC juce::juce_recommended_config_flags)
option(VOCALIS_BUILD_TESTS "Build DSP and plugin verification tools" ON)
if(VOCALIS_BUILD_TESTS)
 enable_testing()
 add_executable(VoiceTests)
 target_sources(VoiceTests PRIVATE tests/VoiceTests.cpp)
 target_link_libraries(VoiceTests PRIVATE VocalisVoice)
 target_include_directories(VoiceTests PRIVATE "${JUCE_PATH}/modules")
 add_test(NAME voice_tests COMMAND VoiceTests)
 add_executable(DspTests tests/DspTests.cpp)
 target_include_directories(DspTests PRIVATE src)
 target_compile_options(DspTests PRIVATE -Wall -Wextra -Wpedantic)
 add_test(NAME dsp_tests COMMAND DspTests)
 juce_add_console_app(Vst3Probe PRODUCT_NAME "Vocalis VST3 Verification")
 target_sources(Vst3Probe PRIVATE tests/Vst3Probe.cpp tests/CaptureX11.cpp)
 target_compile_definitions(Vst3Probe PRIVATE JUCE_MODAL_LOOPS_PERMITTED=1 JUCE_PLUGINHOST_VST3=1 JUCE_PLUGINHOST_VST=0 JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0)
 target_link_libraries(Vst3Probe PRIVATE X11 juce::juce_audio_utils juce::juce_recommended_config_flags)
 add_executable(GuiTests tests/GuiTests.cpp)
 target_include_directories(GuiTests PRIVATE "${JUCE_PATH}/modules")
 target_link_libraries(GuiTests PRIVATE VocalisVoice)
 add_test(NAME gui_tests COMMAND GuiTests "${CMAKE_CURRENT_SOURCE_DIR}/artifacts")
 add_executable(AudioRegression tests/AudioRegression.cpp)
 target_include_directories(AudioRegression PRIVATE "${JUCE_PATH}/modules")
 target_link_libraries(AudioRegression PRIVATE VocalisVoice)
 add_executable(DesignParity tests/DesignParity.cpp)
 target_include_directories(DesignParity PRIVATE "${JUCE_PATH}/modules")
 target_link_libraries(DesignParity PRIVATE VocalisVoice)
 add_test(NAME design_parity COMMAND DesignParity)
 add_executable(AuditDspTests tests/AuditDspTests.cpp)
 target_include_directories(AuditDspTests PRIVATE src)
 add_test(NAME audit_dsp_tests COMMAND AuditDspTests)
 add_executable(ThreadStress tests/ThreadStress.cpp)
 target_include_directories(ThreadStress PRIVATE "${JUCE_PATH}/modules")
 target_link_libraries(ThreadStress PRIVATE VocalisVoice)
 add_test(NAME thread_stress COMMAND ThreadStress)
 add_executable(CallbackAllocation tests/CallbackAllocation.cpp)
 target_include_directories(CallbackAllocation PRIVATE "${JUCE_PATH}/modules")
 target_link_libraries(CallbackAllocation PRIVATE VocalisVoice)
 target_link_options(CallbackAllocation PRIVATE -Wl,--wrap=malloc -Wl,--wrap=calloc -Wl,--wrap=realloc)
 add_test(NAME callback_allocation COMMAND CallbackAllocation)
endif()
